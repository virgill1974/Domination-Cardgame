//**************************************************************
//**************************************************************
// Funktion: EAN Code Auslesen und Auswerten
//**************************************************************
//**************************************************************

unsigned char ean_lesen(void)                                        // Leseroutine
{   
unsigned char temp,kartendaten[3], kartennr, transition, flag; 
unsigned int loop, loop2;
float z[12],zg[3]; 

//**************************************************************

again:

flag = 0;
while (sensor==0)
{
    if (cancel==0) 
    {
        flag=1;
        goto ende;
    }
}                         // Warten auf Karte 

//**************************************************************        
for (loop=0;loop<12;loop++) 
        z[loop]=0;                                                // Initialisierungen

for (loop=0;loop<2 ;loop++) 
{
        zg[loop]=0;kartendaten[loop]=0;
}            
//**************************************************************
while (sensor==1)
{                                                                // Warten auf balken
        if (cancel==0) 
        {
                flag=1;goto ende;                                // Cancel gedrückt?
        }
} 
//**************************************************************
for (loop=0;loop<12;loop++)
{                                                                // 12 Balkenlängen
    transition=sensor;
    while (transition==sensor)
    {   
        delay_us(5); z[loop]=z[loop]+1; 
        if (z[loop]>1000) goto kartenerror;
    }
}
//**************************************************************
zg[0] = ((z[0]+z[1]+z[2]+z[3])/7);                                // Mittelwerte 
zg[1] = ((z[4]+z[5]+z[6]+z[7])/7);
zg[2] = ((z[8]+z[9]+z[10]+z[11])/7);
//**************************************************************             
for (loop=0;loop<3;loop++)
{                                                                // Balkenanzahl 
    for (loop2=0;loop2<4;loop2++)
    {
//        (unsigned char)temp = (z[loop2+(loop*4)]/zg[loop])+0.5;

    temp = (z[loop2+(loop*4)]/zg[loop])+0.5;
        if (temp >3) goto kartenerror;
        kartendaten[loop]=kartendaten[loop]<<2; 
        kartendaten[loop]=kartendaten[loop]|=temp;
    }
}
//**************************************************************
// EAN Code auswerten
//**************************************************************
for (loop=0;loop<3;loop++)
{                                                                     // 3 Datenblöcke holen                                        
    if (kartendaten[loop]==0b11100101) kartendaten[loop]=0;//3211
    if (kartendaten[loop]==0b10101001) kartendaten[loop]=1;//2221
    if (kartendaten[loop]==0b10011010) kartendaten[loop]=2;//2122
    if (kartendaten[loop]==0b01101010) kartendaten[loop]=3;//1222
    if (kartendaten[loop]==0b01011110) kartendaten[loop]=4;//1132
    if (kartendaten[loop]==0b01101101) kartendaten[loop]=5;//1231
    if (kartendaten[loop]==0b10100110) kartendaten[loop]=6;//2212
    if (kartendaten[loop]==0b01110110) kartendaten[loop]=7;//1312 
    if (kartendaten[loop]==0b01100111) kartendaten[loop]=8;//1213
    if (kartendaten[loop]==0b11010110) kartendaten[loop]=9;//3112
}
//**************************************************************
// Fehlererkennung und Addition der Stellen
//**************************************************************
if ((kartendaten[0]>1) || (kartendaten[1]>9) || (kartendaten[2]>9)) // Fehler?
        goto kartenerror;
  
kartennr = (kartendaten[0]*100)+(kartendaten[1]*10)+kartendaten[2]; // Stellen addieren

goto ende ;
//**************************************************************

kartenerror:

errorled();                                                        // LED toggeln
goto again;

ende:
if (flag ==1) return 255;
else return kartennr;
}  




//**************************************************************          
//**************************************************************
// Funktion: Error LED aufblinken lassen
//**************************************************************
//**************************************************************
                                      
void errorled(void)
{
PORTB=0b11011011;        // nicht das Lautsprecher bit setzen!
delay_ms(500);
PORTB=0b00000000;
}


//**************************************************************
//**************************************************************
// Funktion: Spielernummer aus Kartennummer ermitteln
//**************************************************************
//**************************************************************

unsigned char spielernr_ermitteln(unsigned char karte)
{
unsigned char temp;

// es sind 40 versch. Karten in einem Kartenset

if ((karte >= 0) && (karte<=39)) temp=1;
if ((karte >= 40) && (karte<=79)) temp=2;
if ((karte >= 80) && (karte<=119)) temp=3;
if ((karte >= 120) && (karte<=159)) temp=4;

return temp; // Spielernummer
}

//**************************************************************
//**************************************************************
// Funktion: Kartentyp aus Kartennummer ermitteln
//**************************************************************
//**************************************************************

unsigned char kartentyp_ermitteln(unsigned char karte)
{
unsigned char temp;

if (((karte>=00) 
        && (karte<=(gebaeudekarten-1))) 
        || ((karte>=kartenset) 
        && (karte<=(kartenset+gebaeudekarten-1))) 
        || ((karte>=(kartenset*2)) 
        && (karte<=((kartenset*2)+gebaeudekarten-1))) 
        || ((karte>=(kartenset*3)) 
        && (karte<=((kartenset*3)+gebaeudekarten-1)))) 
        temp = 1; // Gebäude

if ((karte>=gebaeudekarten) 
        && (karte<=(gebaeudekarten+einheitenkarten-1))
        || ((karte>=(gebaeudekarten+kartenset)) 
        && (karte<=(kartenset+gebaeudekarten+einheitenkarten-1))) 
        || ((karte>=((kartenset*2)+gebaeudekarten)) 
        && (karte<=((kartenset*2)+gebaeudekarten+einheitenkarten-1))) 
        || ((karte>=((kartenset*3)+gebaeudekarten)) 
        && (karte<=((kartenset*3)+gebaeudekarten+einheitenkarten-1)))) 
        temp = 2; // Einheit

if (((karte>=(gebaeudekarten+einheitenkarten)) 
        && (karte<=(gebaeudekarten+einheitenkarten+upgradeskarten-1))) 
        || ((karte>=(kartenset+gebaeudekarten+einheitenkarten)) 
        && (karte<=(kartenset+gebaeudekarten+einheitenkarten+upgradeskarten-1))) 
        || ((karte>=((kartenset*2)+gebaeudekarten+einheitenkarten)) 
        && (karte<=((kartenset*2)+gebaeudekarten+einheitenkarten+upgradeskarten-1))) 
        || ((karte>=((kartenset*3)+gebaeudekarten+einheitenkarten)) 
        && (karte<=((kartenset*3)+gebaeudekarten+einheitenkarten+upgradeskarten-1)))) 
        temp = 3; // Upgrade

return temp;
}


//**************************************************************
//**************************************************************
// Funktion: Karte löschen und werte ausgleichen
//**************************************************************
//**************************************************************

void karte_loeschen(unsigned char kartee, unsigned char spielerr, unsigned char indexx)
{
unsigned char tempi,typ;

tempi=kartentyp_ermitteln(kartee);
typ = spielernr_ermitteln(kartee);

// Wenn Kraftwerk, dann Energie --
if ((kartenzeiger[kartee].zeiger==kraftwerkusa)        
|| (kartenzeiger[kartee].zeiger==kraftwerkbiotec)
|| (kartenzeiger[kartee].zeiger==kraftwerkchina)) 

{
    if (player[spielerr].energieflag==0)
        player[spielerr].energie=player[spielerr].energie-defaultenergie;                

    if (player[spielerr].energieflag==1)
        player[spielerr].energie=player[spielerr].energie-defaultenergie-2;                      
}


if ((player[spielerr].energie<0)&&(typ!=3))                 // Überlast!
{        
        player[spielerr].spieler_karten[indexx].defensive=3; // defpunkte wiederherstellen
                goto endfunction;                                
}

player[spielerr].spieler_karten[indexx].ean_nummer=255;
player[spielerr].spieler_karten[indexx].verbleibende_runden=0;
player[spielerr].spieler_karten[indexx].freischaltung=0;
player[spielerr].spieler_karten[indexx].defensive=0;
player[spielerr].spieler_karten[indexx].flag=0;

// kein Kraftwerk:
if ((tempi==1) 
        && (kartenzeiger[kartee].zeiger!=kraftwerkusa)
        && (kartenzeiger[kartee].zeiger!=kraftwerkbiotec) 
        && (kartenzeiger[kartee].zeiger!=kraftwerkchina))
        player[spielerr].energie++;        

if (tempi == 1) 
        player[spielerr].anzahl_gebaeude--;

if (tempi == 2) 
        player[spielerr].anzahl_einheiten--;

endfunction:
}

//**************************************************************
//**************************************************************
// Funktion: Zufallszahl von 1 - 6 berechnen
//**************************************************************
//**************************************************************

unsigned char zufallszahl(void)
{

int random;
unsigned char ausgabe;
float testrandom;

random = rand();
testrandom = random / 5461.1667; 
    if (testrandom <=1.0)
        ausgabe = 1;

    else if (testrandom >1.0 && testrandom <= 2.0)  
        ausgabe = 2;

    else if (testrandom >2.0 && testrandom <= 3.0)  
        ausgabe = 3;   

    else if (testrandom >3.0 && testrandom <= 4.0)  
        ausgabe = 4;

    else if (testrandom >4.0 && testrandom <= 5.0)  
        ausgabe = 5;       

    else 
        ausgabe = 6;


return ausgabe;
}
